let completedSteps=new Set([1,2,3,4,5]);
let completedPhase=1;
function loadPhaseState(){
  completedSteps=new Set([1,2,3,4,5]);
  try{
    const raw=localStorage.getItem(phaseStorageKey);
    if(raw!==null){
      const ids=JSON.parse(raw);
      if(Array.isArray(ids))ids.filter(n=>Number.isInteger(n)&&n>=6&&n<=15).forEach(n=>completedSteps.add(n));
    }else{
      const n=JSON.parse(localStorage.getItem(legacyPhaseStorageKey));
      if(Number.isInteger(n)&&n>=1&&n<=5)processPhases.slice(0,n).forEach(p=>p.steps.forEach(s=>completedSteps.add(s.n)));
    }
  }catch{}
  syncPhaseStatuses();
}
function reviewReady(){return [...docsA,...docsB].every(d=>d.status==='approved'||d.status==='submitted');}
function phaseControl(index,inner=false){
  const p=processPhases[index],number=index+1;
  if(number===1)return '';
  const count=p.steps.filter(s=>completedSteps.has(s.n)).length;
  const done=count===p.steps.length,active=number===completedPhase+1;
  let content=`<span class="pill ${done?'green':'blue'}">세부 업무 ${count} / ${p.steps.length} 완료</span>`;
  if(number===2&&!done&&active&&currentRole==='admin')content+=`<button class="primary" onclick="completePhase(2)" ${reviewReady()?'':'disabled'}>서류 검토 승인 및 완료</button>`;
  if(!done)content+=`<small>${currentRole==='admin'?'각 업무의 강제 완료 버튼으로 해당 업무만 완료할 수 있습니다.':'올라 관리자가 세부 업무를 모두 완료하면 다음 단계로 넘어갑니다.'}</small>`;
  if(done&&number<=completedPhase&&number<5)content+=`<button class="secondary" onclick="goToPhase(${number+1})">다음 단계로 이동 →</button>`;
  return inner?content:`<div class="phase-control" ${number===2?'id="phaseReviewControl"':''}>${content}</div>`;
}
function saveCompletedSteps(next){

  completedSteps=next;
  return true;
}
function completeStep(number){
  if(currentRole!=='admin')return toast('완료 처리는 올라 관리자만 가능합니다.');
  const step=processPhases.flatMap(p=>p.steps).find(s=>s.n===number);
  if(!step||number<6||completedSteps.has(number))return;
  const next=new Set(completedSteps);next.add(number);
  if(!saveCompletedSteps(next))return;
  renderProcess();
  toast(step.task+' · 완료 처리되었습니다.');
}
function cancelStep(number){
  if(currentRole!=='admin')return toast('완료 취소는 올라 관리자만 가능합니다.');
  const step=processPhases.flatMap(p=>p.steps).find(s=>s.n===number);
  if(!step||number<6||!completedSteps.has(number))return;
  const next=new Set(completedSteps);next.delete(number);
  if(!saveCompletedSteps(next))return;
  renderProcess();
  toast(step.task+' · 완료가 취소되었습니다.');
}
function completePhase(number){
  if(currentRole!=='admin')return toast('완료 처리는 올라 관리자만 가능합니다.');
  if(number!==2||completedPhase!==1)return toast('각 세부 업무에서 완료 처리해 주세요.');
  if(!reviewReady())return toast('필수서류 제출 및 보완 처리를 먼저 완료해 주세요.');
  const next=new Set(completedSteps);processPhases[1].steps.forEach(s=>next.add(s.n));
  if(!saveCompletedSteps(next))return;

  renderProcess();toast('서류 검토 완료 · 3단계를 진행할 수 있습니다.');
}
function syncPhaseStatuses(){
  completedPhase=0;
  for(const p of processPhases){if(p.steps.every(s=>completedSteps.has(s.n)))completedPhase++;else break;}
  processPhases.forEach((p,i)=>{
    const first=p.steps.find(s=>!completedSteps.has(s.n));
    p.steps.forEach(step=>{step.status=completedSteps.has(step.n)?'done':i===completedPhase&&step===first?'current':'pending';});
  });
}
function goToPhase(number){

  document.getElementById('phase-'+number)?.scrollIntoView({behavior:'smooth',block:'start'});
}

const skuStatusFields=['art','coa','ing','cfs'];
function loadSkuEdits(){
  try{
    const saved=JSON.parse(localStorage.getItem(skuStorageKey)||'[]');
    if(!Array.isArray(saved))return;
    saved.forEach(entry=>{
      if(!entry||typeof entry!=='object')return;
      const row=skuRows.find(r=>r.owner===entry.owner&&r.n===entry.n);if(!row)return;
      [...Object.keys(skuEditFields),'updatedBy','updatedAt'].forEach(key=>{
        if(skuStatusFields.includes(key)?[0,1].includes(entry[key]):typeof entry[key]==='string')row[key]=entry[key];
      });
    });
  }catch{}
}
function canEditSku(row){return row&&(currentRole==='admin'||row.owner==='somebymi');}
function openSkuEdit(owner,n){
  const row=skuRows.find(r=>r.owner===owner&&r.n===n);
  if(!canEditSku(row))return toast('수정 권한이 없는 SKU입니다.');
  editingSku=row;
  document.getElementById('skuEditTitle').textContent=row.name;
  Object.entries(skuEditFields).forEach(([key,id])=>{const el=document.getElementById(id);el.value=String(row[key]??'');el.readOnly=false;});
  document.getElementById('skuAudit').textContent=`최종 수정 · ${row.updatedAt||'-'} · ${row.updatedBy||'-'}`;
  document.getElementById('skuSaveButton').classList.remove('hidden');
  document.getElementById('skuEditModal').classList.remove('hidden');
}
function closeSkuEdit(){document.getElementById('skuEditModal').classList.add('hidden');editingSku=null}
function saveSkuEdit(){
  if(!canEditSku(editingSku))return toast('수정 권한이 없는 SKU입니다.');
  const next={...editingSku};
  for(const [key,id] of Object.entries(skuEditFields)){
    const value=document.getElementById(id).value.trim();
    if(skuStatusFields.includes(key)){
      if(!['0','1'].includes(value))return toast('서류 상태를 선택해 주세요.');
      next[key]=Number(value);
    }else next[key]=value;
  }
  if(!next.name||!next.sku)return toast('제품명과 용량 / SKU를 입력해 주세요.');
  next.updatedBy=currentRole==='admin'?'올라 관리자':'SOME BY MI 담당자';
  next.updatedAt=new Date().toLocaleString('ko-KR');
  try{
    const rows=skuRows.map(r=>r===editingSku?next:r);
    localStorage.setItem(skuStorageKey,JSON.stringify(rows));
  }catch{return toast('저장하지 못했습니다. 브라우저 저장소 설정을 확인해 주세요.');}
  Object.assign(editingSku,next);closeSkuEdit();renderSkuTable();toast('SKU 정보와 서류 상태가 저장되었습니다.');
}
